@mcp.tool()
def prepare_task_payload(assignee_name: str, title: str):
    # Ажилтны ID-г нэрээр нь хайж олох
    user = db.query("SELECT id FROM users WHERE name LIKE %s", f"%{assignee_name}%")
    
    return {
        "action": "OPEN_MODAL",
        "module": "tasks",
        "fields": { "title": title, "assignee_id": user.id }
    }
@app.post("/api/ai/chat")
async def process_ai_prompt(req: PromptRequest):
    response = ai_client.models.generate_content(
        model='gemini-2.5-flash',
        contents=req.prompt,
        config=GenerateContentConfig(tools=[mcp_tools])
    )
    # Gemini MCP хэрэгслийг ашиглан Action буцаана
    return response.function_calls[0].execute()
$.post('/api/ai/chat', { prompt: userPrompt }, function(res) {
    if (res.action === "OPEN_MODAL") {
        $('#' + res.module + 'Modal').modal('show');
        Object.keys(res.fields).forEach(key => {
            $('[name=' + key + ']').val(res.fields[key]);
        });
    }
});
